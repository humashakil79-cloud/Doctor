<script>
    let appointments = [];
    let serialCounter = 1;

    // Load appointments from memory on page load
    function loadAppointments() {
        const saved = localStorage.getItem('appointments');
        const savedCounter = localStorage.getItem('serialCounter');
        
        if (saved) {
            appointments = JSON.parse(saved);
        }
        if (savedCounter) {
            serialCounter = parseInt(savedCounter);
        }
        renderAppointments();
    }

    // Save appointments to memory
    function saveAppointments() {
        localStorage.setItem('appointments', JSON.stringify(appointments));
        localStorage.setItem('serialCounter', serialCounter.toString());
    }

    // Set minimum date to today
    document.getElementById('date').min = new Date().toISOString().split('T')[0];

    document.getElementById('appointmentForm').addEventListener('submit', function(e) {
        e.preventDefault();

        const appointment = {
            id: Date.now(),
            serial: serialCounter++,
            name: document.getElementById('patientName').value,
            phone: document.getElementById('phone').value,
            date: document.getElementById('date').value,
            time: document.getElementById('time').value,
            reason: document.getElementById('reason').value
        };

        appointments.push(appointment);
        saveAppointments();
        renderAppointments();

        // Reset form
        this.reset();
        
        alert(`Appointment booked! Serial Number: ${appointment.serial}`);
    });

    function renderAppointments() {
        const listContainer = document.getElementById('appointmentList');

        if (appointments.length === 0) {
            listContainer.innerHTML = '<div class="empty-state">No appointments scheduled yet</div>';
            return;
        }

        listContainer.innerHTML = appointments.map(apt => `
            <div class="appointment-item">
                <div class="serial-number">${apt.serial}</div>
                <div class="appointment-info">
                    <h3>${apt.name}</h3>
                    <p>📞 ${apt.phone}</p>
                    <p>📅 ${apt.date} at ${apt.time}</p>
                    ${apt.reason ? `<p>💬 ${apt.reason}</p>` : ''}
                </div>
                <button class="delete-btn" onclick="deleteAppointment(${apt.id})">Delete</button>
            </div>
        `).join('');
    }

    function deleteAppointment(id) {
        if (confirm('Are you sure you want to delete this appointment?')) {
            appointments = appointments.filter(apt => apt.id !== id);
            saveAppointments();
            renderAppointments();
        }
    }

    // Load appointments when page loads
    loadAppointments();
</script>
